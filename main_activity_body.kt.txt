class MainActivity : FlutterActivity() {
    private val channelName = "aftabia/native"
    private val pickCode = 4711
    private val pickManyCode = 4712
    private var pendingMany: MethodChannel.Result? = null
    private val sharedUris = ArrayList<Uri>()
    private var pendingPick: MethodChannel.Result? = null
    private val worker = Executors.newSingleThreadExecutor()
    private val ui = Handler(Looper.getMainLooper())

    // These three are only touched from the worker thread.
    private var renderer: PdfRenderer? = null
    private var descriptor: ParcelFileDescriptor? = null
    private var openPath: String? = null

    override fun onCreate(savedInstanceState: android.os.Bundle?) {
        super.onCreate(savedInstanceState)
        collectShared(intent)
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        setIntent(intent)
        collectShared(intent)
    }

    @Suppress("DEPRECATION")
    private fun collectShared(i: Intent?) {
        if (i == null) return
        try {
            val action = i.action
            if (action == Intent.ACTION_SEND) {
                val u = i.getParcelableExtra<Uri>(Intent.EXTRA_STREAM)
                if (u != null) synchronized(sharedUris) { sharedUris.add(u) }
            } else if (action == Intent.ACTION_SEND_MULTIPLE) {
                val list = i.getParcelableArrayListExtra<Uri>(Intent.EXTRA_STREAM)
                if (list != null) synchronized(sharedUris) { sharedUris.addAll(list) }
            }
        } catch (e: Exception) {
        }
    }

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, channelName)
            .setMethodCallHandler { call, result -> handle(call, result) }
    }

    private fun handle(call: MethodCall, result: MethodChannel.Result) {
        when (call.method) {
            "dir" -> result.success(filesDir.absolutePath)
            "pickPdf" -> pickPdf(result)
            "pickPdfs" -> pickPdfs(result)
            "takeShared" -> takeShared(result)
            "pdfOpen" -> {
                val path = call.argument<String>("path")
                if (path == null) result.error("args", "path missing", null) else pdfOpen(path, result)
            }
            "pdfRender" -> {
                val path = call.argument<String>("path")
                val page = call.argument<Int>("page")
                val width = call.argument<Int>("width")
                if (path == null || page == null || width == null) {
                    result.error("args", "arguments missing", null)
                } else {
                    pdfRender(path, page, width, result)
                }
            }
            "viewPdf" -> {
                val path = call.argument<String>("path")
                val name = call.argument<String>("name")
                if (path == null || name == null) {
                    result.error("args", "arguments missing", null)
                } else {
                    viewPdf(path, name, result)
                }
            }
            "pdfClose" -> {
                worker.execute { closeRenderer() }
                result.success(true)
            }
            "saveToDownloads" -> {
                val path = call.argument<String>("path")
                val name = call.argument<String>("name")
                if (path == null || name == null) {
                    result.error("args", "arguments missing", null)
                } else {
                    saveToDownloads(path, name, result)
                }
            }
            "shareText" -> {
                try {
                    val text = call.argument<String>("text") ?: ""
                    val title = call.argument<String>("title") ?: ""
                    val send = Intent(Intent.ACTION_SEND)
                    send.type = "text/plain"
                    send.putExtra(Intent.EXTRA_TEXT, text)
                    startActivity(Intent.createChooser(send, title))
                    result.success(true)
                } catch (e: Exception) {
                    result.error("share", e.message ?: "share failed", null)
                }
            }
            "openUrl" -> {
                try {
                    val url = call.argument<String>("url") ?: ""
                    startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(url)))
                    result.success(true)
                } catch (e: Exception) {
                    result.error("url", e.message ?: "open failed", null)
                }
            }
            "getLocation" -> getLocation(result)
            "compassStart" -> {
                val lat = call.argument<Double>("lat") ?: 0.0
                val lon = call.argument<Double>("lon") ?: 0.0
                result.success(compassStart(lat, lon))
            }
            "compassStop" -> {
                compassStop()
                result.success(true)
            }
            "compassRead" -> result.success(compassRead())
            else -> result.notImplemented()
        }
    }

    // ------------------------------------------------------------
    //  PDF picker (system file chooser)
    // ------------------------------------------------------------

    private fun pickPdf(result: MethodChannel.Result) {
        if (pendingPick != null) {
            result.error("busy", "picker already open", null)
            return
        }
        pendingPick = result
        try {
            val intent = Intent(Intent.ACTION_OPEN_DOCUMENT)
            intent.addCategory(Intent.CATEGORY_OPENABLE)
            intent.type = "application/pdf"
            startActivityForResult(intent, pickCode)
        } catch (e: Exception) {
            pendingPick = null
            result.error("pick", e.message ?: "cannot open picker", null)
        }
    }

    private fun displayName(uri: Uri): String {
        var name = "book.pdf"
        try {
            val cursor = contentResolver.query(uri, null, null, null, null)
            if (cursor != null) {
                try {
                    val i = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                    if (i >= 0 && cursor.moveToFirst()) {
                        val n = cursor.getString(i)
                        if (n != null) name = n
                    }
                } finally {
                    cursor.close()
                }
            }
        } catch (e: Exception) {
        }
        return name
    }

    private fun copyUris(uris: List<Uri>, result: MethodChannel.Result) {
        worker.execute {
            val out = ArrayList<HashMap<String, Any>>()
            var n = 0
            for (uri in uris) {
                try {
                    val name = displayName(uri)
                    val dest = File(cacheDir, "pick_" + System.currentTimeMillis() + "_" + n + ".pdf")
                    n++
                    val input = contentResolver.openInputStream(uri) ?: continue
                    input.use { src ->
                        FileOutputStream(dest).use { o -> src.copyTo(o) }
                    }
                    val map = HashMap<String, Any>()
                    map["path"] = dest.absolutePath
                    map["name"] = name
                    map["size"] = dest.length()
                    out.add(map)
                } catch (e: Exception) {
                }
            }
            ui.post { result.success(out) }
        }
    }

    private fun pickPdfs(result: MethodChannel.Result) {
        if (pendingMany != null) {
            result.error("busy", "picker already open", null)
            return
        }
        pendingMany = result
        try {
            val intent = Intent(Intent.ACTION_OPEN_DOCUMENT)
            intent.addCategory(Intent.CATEGORY_OPENABLE)
            intent.type = "application/pdf"
            intent.putExtra(Intent.EXTRA_ALLOW_MULTIPLE, true)
            startActivityForResult(intent, pickManyCode)
        } catch (e: Exception) {
            pendingMany = null
            result.error("pick", e.message ?: "cannot open picker", null)
        }
    }

    private fun takeShared(result: MethodChannel.Result) {
        val list = ArrayList<Uri>()
        synchronized(sharedUris) {
            list.addAll(sharedUris)
            sharedUris.clear()
        }
        if (list.isEmpty()) {
            result.success(ArrayList<HashMap<String, Any>>())
        } else {
            copyUris(list, result)
        }
    }

    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        if (requestCode == pickManyCode) {
            val r = pendingMany ?: return
            pendingMany = null
            if (resultCode != Activity.RESULT_OK || data == null) {
                r.success(ArrayList<HashMap<String, Any>>())
                return
            }
            val uris = ArrayList<Uri>()
            val clip = data.clipData
            if (clip != null) {
                for (i in 0 until clip.itemCount) {
                    val u = clip.getItemAt(i).uri
                    if (u != null) uris.add(u)
                }
            } else if (data.data != null) {
                uris.add(data.data!!)
            }
            copyUris(uris, r)
            return
        }
        if (requestCode != pickCode) {
            super.onActivityResult(requestCode, resultCode, data)
            return
        }
        val r = pendingPick ?: return
        pendingPick = null
        val uri = data?.data
        if (resultCode != Activity.RESULT_OK || uri == null) {
            r.success(null)
            return
        }
        worker.execute {
            try {
                var name = "book.pdf"
                val cursor = contentResolver.query(uri, null, null, null, null)
                if (cursor != null) {
                    try {
                        val i = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                        if (i >= 0 && cursor.moveToFirst()) {
                            val n = cursor.getString(i)
                            if (n != null) name = n
                        }
                    } finally {
                        cursor.close()
                    }
                }
                val dest = File(cacheDir, "pick_" + System.currentTimeMillis() + ".pdf")
                val input = contentResolver.openInputStream(uri)
                    ?: throw Exception("cannot read the selected file")
                input.use { src ->
                    FileOutputStream(dest).use { out -> src.copyTo(out) }
                }
                val map = HashMap<String, Any>()
                map["path"] = dest.absolutePath
                map["name"] = name
                map["size"] = dest.length()
                ui.post { r.success(map) }
            } catch (e: Exception) {
                ui.post { r.error("pick", e.message ?: "pick failed", null) }
            }
        }
    }

    // ------------------------------------------------------------
    //  PDF rendering with the built-in Android PdfRenderer
    // ------------------------------------------------------------

    private fun closeRenderer() {
        try {
            renderer?.close()
        } catch (e: Exception) {
        }
        try {
            descriptor?.close()
        } catch (e: Exception) {
        }
        renderer = null
        descriptor = null
        openPath = null
    }

    private fun ensureOpen(path: String) {
        if (renderer != null && openPath == path) return
        closeRenderer()
        var fd: ParcelFileDescriptor? = null
        try {
            fd = ParcelFileDescriptor.open(File(path), ParcelFileDescriptor.MODE_READ_ONLY)
            val r = PdfRenderer(fd)
            descriptor = fd
            renderer = r
            openPath = path
        } catch (e: Exception) {
            try {
                fd?.close()
            } catch (e2: Exception) {
            }
            throw e
        }
    }

    private fun pdfOpen(path: String, result: MethodChannel.Result) {
        worker.execute {
            try {
                ensureOpen(path)
                val r = renderer!!
                val info = ArrayList<Int>()
                info.add(r.pageCount)
                if (r.pageCount > 0) {
                    val p = r.openPage(0)
                    info.add(p.width)
                    info.add(p.height)
                    p.close()
                } else {
                    info.add(0)
                    info.add(0)
                }
                ui.post { result.success(info) }
            } catch (e: Exception) {
                ui.post { result.error("open", e.message ?: "open failed", null) }
            }
        }
    }

    private fun looksBlank(bmp: Bitmap): Boolean {
        val w = bmp.width
        val h = bmp.height
        val stepX = if (w / 150 < 1) 1 else w / 150
        val stepY = if (h / 220 < 1) 1 else h / 220
        var found = 0
        var y = 0
        while (y < h) {
            var x = 0
            while (x < w) {
                val c = bmp.getPixel(x, y)
                if ((c and 0x00FFFFFF) != 0x00FFFFFF) {
                    found++
                    if (found >= 3) return false
                }
                x += stepX
            }
            y += stepY
        }
        return true
    }

    private fun pdfRender(path: String, index: Int, width: Int, result: MethodChannel.Result) {
        worker.execute {
            try {
                ensureOpen(path)
                val r = renderer!!
                if (index < 0 || index >= r.pageCount) {
                    throw Exception("page out of range")
                }
                val page = r.openPage(index)
                var blank = false
                var bytes = ByteArray(0)
                val w = if (width < 200) 200 else width
                var h = 0
                try {
                    h = (w.toLong() * page.height.toLong() / page.width.toLong()).toInt()
                    if (h < 1) h = 1
                    val bmp = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888)
                    bmp.eraseColor(Color.WHITE)
                    page.render(bmp, null, null, PdfRenderer.Page.RENDER_MODE_FOR_DISPLAY)
                    if (looksBlank(bmp)) {
                        bmp.eraseColor(Color.WHITE)
                        page.render(bmp, null, null, PdfRenderer.Page.RENDER_MODE_FOR_PRINT)
                        blank = looksBlank(bmp)
                    }
                    val out = ByteArrayOutputStream()
                    bmp.compress(Bitmap.CompressFormat.PNG, 100, out)
                    bmp.recycle()
                    bytes = out.toByteArray()
                } finally {
                    page.close()
                }
                val map = HashMap<String, Any>()
                map["bytes"] = bytes
                map["w"] = w
                map["h"] = h
                map["blank"] = blank
                ui.post { result.success(map) }
            } catch (e: Throwable) {
                ui.post { result.error("render", e.message ?: "render failed", null) }
            }
        }
    }

    // ------------------------------------------------------------
    //  Open the PDF in another PDF viewer app on the phone
    // ------------------------------------------------------------

    private fun viewPdf(path: String, name: String, result: MethodChannel.Result) {
        worker.execute {
            try {
                if (Build.VERSION.SDK_INT < Build.VERSION_CODES.Q) {
                    throw Exception("needs Android 10 or newer")
                }
                val src = File(path)
                val values = ContentValues()
                values.put(MediaStore.MediaColumns.DISPLAY_NAME, name)
                values.put(MediaStore.MediaColumns.MIME_TYPE, "application/pdf")
                values.put(MediaStore.MediaColumns.RELATIVE_PATH, Environment.DIRECTORY_DOWNLOADS)
                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
                    ?: throw Exception("cannot create the file")
                val out = contentResolver.openOutputStream(uri)
                    ?: throw Exception("cannot write the file")
                out.use { o ->
                    src.inputStream().use { input -> input.copyTo(o) }
                }
                ui.post {
                    try {
                        val view = Intent(Intent.ACTION_VIEW)
                        view.setDataAndType(uri, "application/pdf")
                        view.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
                        startActivity(Intent.createChooser(view, name))
                        result.success(true)
                    } catch (e: Exception) {
                        result.error("view", e.message ?: "no viewer", null)
                    }
                }
            } catch (e: Exception) {
                ui.post { result.error("view", e.message ?: "view failed", null) }
            }
        }
    }

    // ------------------------------------------------------------
    //  Save a copy into the phone's Downloads folder
    // ------------------------------------------------------------

    private fun saveToDownloads(path: String, name: String, result: MethodChannel.Result) {
        worker.execute {
            try {
                val src = File(path)
                val shown: String
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                    val values = ContentValues()
                    values.put(MediaStore.MediaColumns.DISPLAY_NAME, name)
                    values.put(MediaStore.MediaColumns.MIME_TYPE, "application/pdf")
                    values.put(MediaStore.MediaColumns.RELATIVE_PATH, Environment.DIRECTORY_DOWNLOADS)
                    val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
                        ?: throw Exception("cannot create the file")
                    val out = contentResolver.openOutputStream(uri)
                        ?: throw Exception("cannot write the file")
                    out.use { o ->
                        src.inputStream().use { input -> input.copyTo(o) }
                    }
                    shown = "Downloads/" + name
                } else {
                    val dir = getExternalFilesDir(Environment.DIRECTORY_DOWNLOADS) ?: filesDir
                    dir.mkdirs()
                    val dest = File(dir, name)
                    src.inputStream().use { input ->
                        FileOutputStream(dest).use { o -> input.copyTo(o) }
                    }
                    shown = dest.absolutePath
                }
                ui.post { result.success(shown) }
            } catch (e: Exception) {
                ui.post { result.error("save", e.message ?: "save failed", null) }
            }
        }
    }

    // ------------------------------------------------------------
    //  Location (one-shot, permission asked at the moment of use)
    // ------------------------------------------------------------

    private val locCode = 4712
    private var pendingLoc: MethodChannel.Result? = null
    private var locListener: LocationListener? = null
    private var locBest: Location? = null
    private val locTimeout = Runnable { finishLocation("timeout") }

    private fun hasLocPermission(): Boolean {
        if (Build.VERSION.SDK_INT < 23) return true
        return checkSelfPermission(Manifest.permission.ACCESS_FINE_LOCATION) == PackageManager.PERMISSION_GRANTED ||
            checkSelfPermission(Manifest.permission.ACCESS_COARSE_LOCATION) == PackageManager.PERMISSION_GRANTED
    }

    private fun getLocation(result: MethodChannel.Result) {
        if (pendingLoc != null) {
            result.error("busy", "location request already running", null)
            return
        }
        pendingLoc = result
        locBest = null
        if (!hasLocPermission()) {
            if (Build.VERSION.SDK_INT >= 23) {
                requestPermissions(
                    arrayOf(Manifest.permission.ACCESS_FINE_LOCATION, Manifest.permission.ACCESS_COARSE_LOCATION),
                    locCode
                )
            } else {
                finishLocation("denied")
            }
            return
        }
        startLocation()
    }

    override fun onRequestPermissionsResult(requestCode: Int, permissions: Array<out String>, grantResults: IntArray) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)
        if (requestCode != locCode) return
        if (pendingLoc == null) return
        if (hasLocPermission()) startLocation() else finishLocation("denied")
    }

    private fun startLocation() {
        try {
            val lm = getSystemService(Context.LOCATION_SERVICE) as LocationManager
            var best: Location? = null
            for (p in lm.getProviders(true)) {
                try {
                    val l = lm.getLastKnownLocation(p)
                    if (l != null && (best == null || l.time > best.time)) best = l
                } catch (e: SecurityException) {
                } catch (e: Exception) {
                }
            }
            locBest = best
            val fresh = best != null && (System.currentTimeMillis() - best.time) < 30L * 60L * 1000L
            if (fresh) {
                finishLocation("ok")
                return
            }
            val providers = ArrayList<String>()
            if (lm.isProviderEnabled(LocationManager.GPS_PROVIDER)) providers.add(LocationManager.GPS_PROVIDER)
            if (lm.isProviderEnabled(LocationManager.NETWORK_PROVIDER)) providers.add(LocationManager.NETWORK_PROVIDER)
            if (providers.isEmpty()) {
                finishLocation(if (best != null) "ok" else "off")
                return
            }
            val listener = object : LocationListener {
                override fun onLocationChanged(location: Location) {
                    locBest = location
                    finishLocation("ok")
                }

                override fun onStatusChanged(provider: String?, status: Int, extras: android.os.Bundle?) {}
                override fun onProviderEnabled(provider: String) {}
                override fun onProviderDisabled(provider: String) {}
            }
            locListener = listener
            for (p in providers) {
                lm.requestLocationUpdates(p, 0L, 0f, listener, Looper.getMainLooper())
            }
            ui.postDelayed(locTimeout, 25000L)
        } catch (e: SecurityException) {
            finishLocation("denied")
        } catch (e: Exception) {
            finishLocation("timeout")
        }
    }

    private fun finishLocation(status: String) {
        val r = pendingLoc ?: return
        pendingLoc = null
        ui.removeCallbacks(locTimeout)
        try {
            val l = locListener
            if (l != null) {
                val lm = getSystemService(Context.LOCATION_SERVICE) as LocationManager
                lm.removeUpdates(l)
            }
        } catch (e: Exception) {
        }
        locListener = null
        val map = HashMap<String, Any>()
        val best = locBest
        if (best != null && (status == "ok" || status == "timeout")) {
            map["status"] = "ok"
            map["lat"] = best.latitude
            map["lon"] = best.longitude
        } else {
            map["status"] = status
        }
        r.success(map)
    }

    // ------------------------------------------------------------
    //  Compass: rotation-vector sensor (gyro + accelerometer + magnetometer
    //  fused by Android), works flat on a table and when held upright.
    // ------------------------------------------------------------

    private var sensorMgr: SensorManager? = null
    private var compassOn = false
    private var usingRotationVector = false
    private val rot = FloatArray(9)
    private val accelV = FloatArray(3)
    private val magV = FloatArray(3)
    private var haveAccel = false
    private var haveMag = false
    private var headingDeg = -1.0
    private var flatMode = true
    private var sensorAccuracy = 0
    private var declination = 0f

    private val sensorListener = object : SensorEventListener {
        override fun onSensorChanged(e: SensorEvent) {
            when (e.sensor.type) {
                Sensor.TYPE_ROTATION_VECTOR -> {
                    SensorManager.getRotationMatrixFromVector(rot, e.values)
                    computeHeading()
                }
                Sensor.TYPE_ACCELEROMETER -> {
                    if (!usingRotationVector) {
                        for (i in 0..2) accelV[i] = accelV[i] * 0.8f + e.values[i] * 0.2f
                        haveAccel = true
                        computeFromRaw()
                    }
                }
                Sensor.TYPE_MAGNETIC_FIELD -> {
                    if (!usingRotationVector) {
                        for (i in 0..2) magV[i] = magV[i] * 0.8f + e.values[i] * 0.2f
                        haveMag = true
                        computeFromRaw()
                    }
                }
            }
        }

        override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {
            if (sensor != null && sensor.type == Sensor.TYPE_MAGNETIC_FIELD) sensorAccuracy = accuracy
        }
    }

    private fun computeFromRaw() {
        if (!haveAccel || !haveMag) return
        if (SensorManager.getRotationMatrix(rot, null, accelV, magV)) computeHeading()
    }

    private fun computeHeading() {
        // rot maps phone axes to world (East, North, Up).
        // Flat on a table: use the top edge of the phone. Upright: use the back camera direction.
        val flat = Math.abs(rot[8]) > 0.6
        val az = if (flat) Math.atan2(rot[1].toDouble(), rot[4].toDouble())
        else Math.atan2((-rot[2]).toDouble(), (-rot[5]).toDouble())
        var deg = Math.toDegrees(az) + declination.toDouble()
        deg = ((deg % 360.0) + 360.0) % 360.0
        headingDeg = deg
        flatMode = flat
    }

    private fun compassStart(lat: Double, lon: Double): Boolean {
        compassStop()
        try {
            val sm = getSystemService(Context.SENSOR_SERVICE) as SensorManager
            sensorMgr = sm
            declination = try {
                GeomagneticField(lat.toFloat(), lon.toFloat(), 0f, System.currentTimeMillis()).declination
            } catch (e: Exception) {
                0f
            }
            headingDeg = -1.0
            haveAccel = false
            haveMag = false
            sensorAccuracy = 0
            val rv = sm.getDefaultSensor(Sensor.TYPE_ROTATION_VECTOR)
            val mag = sm.getDefaultSensor(Sensor.TYPE_MAGNETIC_FIELD)
            val acc = sm.getDefaultSensor(Sensor.TYPE_ACCELEROMETER)
            var ok = false
            if (rv != null && mag != null) {
                usingRotationVector = true
                ok = sm.registerListener(sensorListener, rv, SensorManager.SENSOR_DELAY_GAME)
                sm.registerListener(sensorListener, mag, SensorManager.SENSOR_DELAY_UI)
            } else if (mag != null && acc != null) {
                usingRotationVector = false
                ok = sm.registerListener(sensorListener, acc, SensorManager.SENSOR_DELAY_GAME)
                sm.registerListener(sensorListener, mag, SensorManager.SENSOR_DELAY_GAME)
            }
            compassOn = ok
            return ok
        } catch (e: Exception) {
            return false
        }
    }

    private fun compassStop() {
        try {
            sensorMgr?.unregisterListener(sensorListener)
        } catch (e: Exception) {
        }
        compassOn = false
    }

    private fun compassRead(): HashMap<String, Any>? {
        if (!compassOn || headingDeg < 0.0) return null
        val m = HashMap<String, Any>()
        m["heading"] = headingDeg
        m["accuracy"] = sensorAccuracy
        m["flat"] = flatMode
        return m
    }

    override fun onDestroy() {
        compassStop()
        super.onDestroy()
    }
}
